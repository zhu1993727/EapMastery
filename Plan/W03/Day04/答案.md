# W03 · Day04 答案与参考实现

## 一、自测题标准答案
1. **为何不可变**：构造后不被多线程改坏，编码结果可缓存（同一消息重发/打日志一致）；W4 primary 超时重发、Spool 队列里消息要保持原样，可变树会让"重发的内容和第一次不一样"，难以排障。
2. **SecsList.Encode**：长度字段写**直接子项个数**；值字节 = 依次调用每个子项 Encode 后拼接；再前置 `List.WriteHeader(子项数)`。
3. **构造即校验值域**：类型选择是建模决策，越早失败越好——构造时抛异常能定位到"谁塞了越界值"，等到编码/发送才报错会远离根因，甚至可能已发出半截错误消息。
4. **300 个 U1**：头部 = 格式字节 1 + 长度 2 字节（300>255）= 3 字节（格式字节变 0xA5，长度 `01 2C`）；值 = 300×1 = 300 字节；共 303 字节。
5. **浮点大端**：先把 float/double 用 `BitConverter.GetBytes` 取主机字节（小端），再 `BinaryPrimitives`/反转成大端；更稳妥是 `BinaryPrimitives.WriteSingleBigEndian/WriteDoubleBigEndian`（.NET 直接支持）。
6. **扩展字符项**：在 SecsFormat 加类型、在"宽度/编码策略"层扩展字符编码（JIS/UTF16），编码器主干只调用"取字节"策略，不需改 List/长度/递归逻辑——这就是把"结构编码"和"值编码"分离的好处。

## 二、参考实现（EapLab.Secs2/Items/）

### SecsItem 抽象与 List/Ascii/Boolean/Bytes
```csharp
using EapLab.Secs2.Coding;

namespace EapLab.Secs2.Items;

public abstract class SecsItem
{
    public abstract SecsFormat Format { get; }
    public abstract int Count { get; }
    public abstract byte[] ValueBytes();
    public byte[] Encode()
    {
        var val = ValueBytes();
        var head = Format.WriteHeader(Count);   // List 的 Count=子项数
        var all = new byte[head.Length + val.Length];
        head.CopyTo(all, 0); val.CopyTo(all, head.Length);
        return all;
    }
    public string ToHex() => Convert.ToHexString(Encode());
}

public sealed class SecsList : SecsItem
{
    private readonly SecsItem[] _items;
    public SecsList(params SecsItem[] items) => _items = items;
    public override SecsFormat Format => SecsFormat.List;
    public override int Count => _items.Length;
    public override byte[] ValueBytes()
    {
        var parts = _items.Select(i => i.Encode());
        var all = new byte[parts.Sum(p => p.Length)];
        int o = 0; foreach (var p in parts) { p.CopyTo(all, o); o += p.Length; }
        return all;
    }
}

public sealed class SecsAscii : SecsItem
{
    private readonly byte[] _b;
    public SecsAscii(string s) => _b = System.Text.Encoding.ASCII.GetBytes(s);
    public override SecsFormat Format => SecsFormat.Ascii;
    public override int Count => _b.Length;
    public override byte[] ValueBytes() => _b;
}

public sealed class SecsBoolean : SecsItem
{
    private readonly byte[] _b;
    public SecsBoolean(params bool[] v) => _b = v.Select(x => (byte)(x ? 1 : 0)).ToArray();
    public override SecsFormat Format => SecsFormat.Boolean;
    public override int Count => _b.Length;
    public override byte[] ValueBytes() => _b;
}

public sealed class SecsBytes : SecsItem   // B
{
    private readonly byte[] _b;
    public SecsBytes(params byte[] b) => _b = b;
    public override SecsFormat Format => SecsFormat.Binary;
    public override int Count => _b.Length;
    public override byte[] ValueBytes() => _b;
}
```

### 数值项（U/I/F 统一，构造即校验）
```csharp
public sealed class SecsNumber : SecsItem
{
    private readonly byte[] _val;
    public override SecsFormat Format { get; }
    public override int Count { get; }
    public SecsNumber(SecsFormat f, params long[] values)   // 整数走这里
    {
        Format = f; Count = values.Length;
        using var ms = new MemoryStream();
        foreach (var v in values) WriteIntBigEndian(ms, f, v);   // 内部按宽度校验范围
        _val = ms.ToArray();
    }
    public SecsNumber(SecsFormat f, params double[] values)     // F4/F8
    {
        Format = f; Count = values.Length;
        Span<byte> b4 = stackalloc byte[4]; Span<byte> b8 = stackalloc byte[8];
        using var ms = new MemoryStream();
        foreach (var v in values)
        {
            if (f == SecsFormat.F4) { System.Buffers.Binary.BinaryPrimitives.WriteSingleBigEndian(b4, (float)v); ms.Write(b4); }
            else { System.Buffers.Binary.BinaryPrimitives.WriteDoubleBigEndian(b8, v); ms.Write(b8); }
        }
        _val = ms.ToArray();
    }
    public override byte[] ValueBytes() => _val;

    private static void WriteIntBigEndian(Stream ms, SecsFormat f, long v)
    {
        bool signed = f is SecsFormat.I1 or SecsFormat.I2 or SecsFormat.I4 or SecsFormat.I8;
        int w = f.AtomWidth();
        (long min, long max) = f switch
        {
            SecsFormat.U1 => (0, byte.MaxValue),
            SecsFormat.U2 => (0, ushort.MaxValue),
            SecsFormat.U4 => (0, uint.MaxValue),
            SecsFormat.I1 => (sbyte.MinValue, sbyte.MaxValue),
            SecsFormat.I2 => (short.MinValue, short.MaxValue),
            SecsFormat.I4 => (int.MinValue, int.MaxValue),
            _ => (long.MinValue, long.MaxValue)
        };
        if (v < min || v > max) throw new ArgumentOutOfRangeException(nameof(v), $"{f} 越界:{v}");
        Span<byte> b = stackalloc byte[8];
        System.Buffers.Binary.BinaryPrimitives.WriteInt64BigEndian(b, v);
        ms.Write(b.Slice(8 - w).ToArray());
    }
}
```

### 静态工厂（贴近标准写法）
```csharp
public static class Secs
{
    public static SecsList L(params SecsItem[] i) => new(i);
    public static SecsAscii A(string s) => new(s);
    public static SecsBoolean Bool(params bool[] v) => new(v);
    public static SecsBytes B(params byte[] v) => new(v);
    public static SecsNumber U1(params long[] v) => new(SecsFormat.U1, v);
    public static SecsNumber U2(params long[] v) => new(SecsFormat.U2, v);
    public static SecsNumber U4(params long[] v) => new(SecsFormat.U4, v);
    public static SecsNumber I1(params long[] v) => new(SecsFormat.I1, v);
    public static SecsNumber I2(params long[] v) => new(SecsFormat.I2, v);
    public static SecsNumber F4(params double[] v) => new(SecsFormat.F4, v);
}
```

public void 例1_L2_A_OK_U1_7()

## 三、今日验收
- 例1/例2/例3 代码输出与 Day03 手算稿逐字节一致。
- 越界值在构造阶段抛异常；空 List/空 A 正确。
- 浮点输出为大端 IEEE754。

## 四、易错点
1. List 用"值字节长度"而不是子项数做 Count。
2. 数值越界拖到发送才暴露。
3. 浮点/多字节整数没转大端。
4. 工厂方法命名和标准类型对不上，读代码时增加心智负担。

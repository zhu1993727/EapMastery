# W11 · Day06 答案与参考实现

## 一、自测题标准答案
1. **为何信号总线区分所有者可写/对方只读**：E84 每根物理信号线只有一个驱动源（设备输出的信号只能由设备 PLC 置位，小车输出的只能由小车置位），对方只能采样。代码上强制"所有者可写、对方只读"是对物理现实的建模：①防止程序 bug 让设备去改本应由小车驱动的 TR_REQ/BUSY，造成"自己给自己发请求"的假握手；②两侧状态机职责清晰，和真实 I/O 点表一一对应，将来接真实 PLC 时只需把内存读写换成 I/O 读写；③便于排查——任何信号异常都能唯一定位到它的驱动方。这和 SECS 的主从、OSS 的单一状态源思想一致：每个状态只有一个权威写者。
2. **VALID 门控实现 / 无效时返回**：总线内部保存原始电平和稳定时间，`TryRead(signal)` 先判断 VALID 是否有效且经过去抖稳定，再判断目标信号是否稳定；VALID 无效或信号未稳定时不返回电平，而是返回"不可信"（用 `bool?`：true/false/null，null=不可信）或抛出/返回带状态的结果。调用方（状态机）只在拿到可信值时才据此迁移，拿到 null 就进入冻结/等待，绝不把不可信信号当真。这样把"VALID 无效不解读、不去抖不采信"在类型层面强制出来。
3. **为何手动步进时钟而非真实线程**：交接握手是严格时序逻辑，用真实线程/Task.Delay 会引入调度不确定性、竞态、难以复现的时序 bug，测试要靠 sleep、又慢又不稳。手动步进时钟（每次 Tick 推进逻辑时间、两个 FSM 用同一时钟各采样推进一次）使整个过程**确定、可复现、可单步、可快进**：给定相同输入和故障注入点，每次跑出完全相同的信号轨迹，精确验证"第几拍、第几毫秒该置什么信号、超时何时触发"，xUnit 无需等待真实时间。这也是工业 PLC 逻辑仿真和协议栈测试的标准做法（逻辑时间与物理时间解耦）。
4. **收到 TR_REQ 后置 ES 前三项检查**：①**方向与物理状态一致性**——本次是 Deposit 还是 Pickup？Deposit 要求口空（HasCarrier=false），Pickup 要求口上有盒（HasCarrier=true），矛盾（空口取/有盒放、方向不明）拒绝告警；②**系统/安全状态**——CS 就绪、无 ESTOP、非 BYPS、VALID 有效、机构无故障；③**逻辑许可（Pickup 尤其需要）**——E87/Host 是否允许下料（LogicPermitsPickup），在制盒没有下料许可不能交出。三项全过，Deposit 直接置 ES，Pickup 还要先把盒送到交接位再置 ES。
5. **COMPT 后传感器判据代码区分**：状态机在 WaitComplete 采样到 COMPT 稳定后，按本次交接方向分支：Deposit 分支要求 `physical.HasCarrier()==true`（在位传感器 ON、坐实）才进入承接/复位，否则进 SafeHold；Pickup 分支要求 `physical.HasCarrier()==false`（在位 OFF、口空）才复位，否则 SafeHold。即同一个 COMPT 信号，完成判据的传感器期望值相反，用一个枚举 TransferDirection 决定断言方向，不符都按"声明与事实冲突"安全保持。
6. **步进模型里超时计时/触发**：进入每个等待态时记录该态的逻辑进入时间戳和该态配置的超时时限（如 WaitComplete 配 T_compt）；每次 Tick 用当前逻辑时间减进入时间，超过时限且退出条件仍未满足就触发超时迁移（按 Day05 矩阵：空闲侧超时回 Idle，动作中超时转 SafeHold）。因为是逻辑时钟，测试可以一次 Tick 一大段时间精确触发超时，也可以断言"超时前 1ms 仍在等待、超时当拍转安全态"。
7. **如何证明没 ES 小车动了也没用**：在模拟器里做故障注入——让 VehicleFsm 不等 ES、直接置 BUSY 执行；然后断言 LoadPortFsm 仍停在等待许可态、不做任何承接/机构动作（因为设备只在自己置了 ES 且握手拍匹配后才会响应 BUSY），并且由于设备不配合、小车侧最终也会因握手不一致/超时进入异常，交接无法完成。再跑一次正常版（先 ES 后 BUSY）对比，两次轨迹并排即可直观证明：物理交接必须双方互锁，单方抢拍推进不了流程。这就是"互锁"的可执行证据，也是给同事讲解的最好 demo。

```csharp
namespace EapLab.E84;

public enum LptSignal
{
    // 设备 -> 小车（设备输出）
    CS, ES, HO0, HO1,
    // 小车 -> 设备（小车输出）
    SS, VR, TR_REQ, BUSY, COMPT,
    // 共用安全/控制
    VALID, ESTOP, BYPS
}
public enum SignalOwner { Equipment, Vehicle, Shared }
public enum TransferDirection { Deposit, Pickup }
```

### 并行信号总线（所有者可写、VALID 门控、去抖、快照）
```csharp
public sealed class LptSignalBus
{
    private readonly Dictionary<LptSignal,bool> _level = new();
    private readonly Dictionary<LptSignal,long> _since = new();
    private readonly Dictionary<LptSignal,SignalOwner> _owner = new();
    private long _now;                       // 逻辑时钟(ms)
    public int DebounceMs { get; init; } = 5;

    public LptSignalBus()
    {
        foreach (LptSignal s in Enum.GetValues<LptSignal>())
        {
            _level[s] = false; _since[s] = 0;
            _owner[s] = s switch
            {
                LptSignal.CS or LptSignal.ES or LptSignal.HO0 or LptSignal.HO1 => SignalOwner.Equipment,
                LptSignal.SS or LptSignal.VR or LptSignal.TR_REQ
                    or LptSignal.BUSY or LptSignal.COMPT => SignalOwner.Vehicle,
                _ => SignalOwner.Shared
            };
        }
    }
    public void Tick(long deltaMs) => _now += deltaMs;

    public void Write(SignalOwner who, LptSignal s, bool on)
    {
        if (_owner[s] != SignalOwner.Shared && _owner[s] != who)
            throw new InvalidOperationException($"信号 {s} 只能由 {_owner[s]} 驱动"); // 唯一写者
        if (_level[s] != on) { _level[s] = on; _since[s] = _now; }
    }
    private bool Stable(LptSignal s) => _level[s] && (_now - _since[s]) >= DebounceMs;

    /// <summary>VALID 门控读取：无效/未稳定返回 null（不可信）。</summary>
    public bool? TryRead(LptSignal s)
    {
        if (!Stable(LptSignal.VALID)) return null;          // VALID 无效→不解读
        if (!_level[s]) return false;
        return Stable(s) ? true : null;                     // 未去抖稳定→不可信
    }
    public bool Raw(LptSignal s) => _level[s];
    public IReadOnlyDictionary<LptSignal,bool> Snapshot()
        => new Dictionary<LptSignal,bool>(_level);
}
```

### 物理口与逻辑许可接口

```csharp
public interface IPortPhysical
{
    bool HasCarrier();                 // 在位传感器
    bool LogicPermitsPickup();         // E87/Host 下料许可
    void DockCarrier();                // 设备承接(Deposit 后)
    void UndockCarrier();              // 设备送出(Pickup 备货/取走后)
    void MoveToHandoverPos();          // Pickup: 把盒送到交接位
}
public sealed class InMemoryPort : IPortPhysical
{
    private bool _has; private bool _pickupPermit = true;
    public bool HasCarrier() => _has;
    public bool LogicPermitsPickup() => _pickupPermit;
    public void SetPickupPermit(bool ok) => _pickupPermit = ok;
    public void DockCarrier() => _has = true;
    public void UndockCarrier() => _has = false;
    public void MoveToHandoverPos() { }
}
```

### 设备侧交接状态机（节选核心）
```csharp
public enum PortState { Idle, WaitRequest, WaitBusy, ObserveTransfer, WaitComplete, Closing,
                        SafeHold, EStopHold, ManualBypass, Rejected }
public sealed class LoadPortFsm
{
    private readonly LptSignalBus _bus; private readonly IPortPhysical _phy;
    public PortState State { get; private set; } = PortState.Idle;
    private long _enterTime; private TransferDirection _dir;
    public List<string> Trace { get; } = new();
    public long T_es=500, T_busy=500, T_compt=30000, T_reset=1000;
    public LoadPortFsm(LptSignalBus bus, IPortPhysical phy){ _bus=bus; _phy=phy; }

    private void Go(PortState s, long now){ State=s; _enterTime=now; Trace.Add($"设备->{s}"); }
    public void SetReady(bool on) => _bus.Write(SignalOwner.Equipment, LptSignal.CS, on);

    public void Tick(long now)
    {
        if (_bus.TryRead(LptSignal.ESTOP) == true) { Go(PortState.EStopHold, now); return; }
        if (_bus.TryRead(LptSignal.BYPS) == true)  { Go(PortState.ManualBypass, now); return; }
        if (_bus.TryRead(LptSignal.VALID) == null) { if (State is not PortState.Idle) Go(PortState.SafeHold, now); return; }

        switch (State)
        {
            case PortState.Idle:
                _bus.Write(SignalOwner.Equipment, LptSignal.ES, false);
                if (_bus.TryRead(LptSignal.TR_REQ) == true) OnRequest(now);
                break;
            case PortState.WaitBusy:
                if (_bus.TryRead(LptSignal.BUSY) == true) Go(PortState.ObserveTransfer, now);
                else if (now-_enterTime > T_busy) { _bus.Write(SignalOwner.Equipment,LptSignal.ES,false); Go(PortState.Idle,now); }
                break;
            case PortState.ObserveTransfer:
                if (_bus.TryRead(LptSignal.COMPT) == true) Go(PortState.WaitComplete, now);
                else if (now-_enterTime > T_compt) Go(PortState.SafeHold, now); // 动作中超时→安全保持
                break;
            case PortState.WaitComplete:
                bool ok = _dir==TransferDirection.Deposit ? _phy.HasCarrier() : !_phy.HasCarrier();
                if (ok) { // 传感器与方向一致→承接/复位
                    if (_dir==TransferDirection.Deposit) _phy.DockCarrier();
                    _bus.Write(SignalOwner.Equipment, LptSignal.ES, false);
                    Go(PortState.Closing, now);
                }
                else Go(PortState.SafeHold, now); // COMPT 与传感器矛盾→安全保持
                break;
            case PortState.Closing:
                if (_bus.TryRead(LptSignal.TR_REQ) == false) Go(PortState.Idle, now);
                else if (now-_enterTime > T_reset) Go(PortState.SafeHold, now);
                break;
        }
    }

    private void OnRequest(long now)
    {
        // 方向由外部调度给定（这里用物理状态推断教学：有盒=来取，空口=来放）
        _dir = _phy.HasCarrier() ? TransferDirection.Pickup : TransferDirection.Deposit;
        bool physicalOk = _dir==TransferDirection.Deposit ? !_phy.HasCarrier() : _phy.HasCarrier();
        bool logicOk = _dir==TransferDirection.Deposit || _phy.LogicPermitsPickup();
        if (!physicalOk || !logicOk) { Go(PortState.Rejected, now); Trace.Add("拒绝:方向/逻辑不符"); return; }
        if (_dir==TransferDirection.Pickup) _phy.MoveToHandoverPos(); // 取货先备货
        _bus.Write(SignalOwner.Equipment, LptSignal.ES, true);        // 置许可
        Go(PortState.WaitBusy, now);
    }
}
```

### 小车侧状态机（节选，故障注入点）
```csharp
public enum VehState { Idle, Approach, Requesting, Acting, Done, Leave }
public sealed class VehicleFsm
{
    private readonly LptSignalBus _bus; private readonly IPortPhysical _phy;
    public VehState State { get; private set; } = VehState.Idle;
    public bool FaultSkipWaitES;       // 故障:不等ES直接动
    public bool FaultNoComplete;       // 故障:不发COMPT(制造T_compt)
    private long _t;
    public VehicleFsm(LptSignalBus bus, IPortPhysical phy){ _bus=bus; _phy=phy; }
    public void Begin(TransferDirection dir, long now)
    {
        _dir=dir; State=VehState.Approach; _t=now;
        _bus.Write(SignalOwner.Vehicle, LptSignal.SS,true);
        _bus.Write(SignalOwner.Vehicle, LptSignal.VR,true);
    }
    private TransferDirection _dir;
    public void Tick(long now)
    {
        switch (State)
        {
            case VehState.Approach:
                _bus.Write(SignalOwner.Vehicle,LptSignal.TR_REQ,true); State=VehState.Requesting; break;
            case VehState.Requesting:
                bool es = _bus.TryRead(LptSignal.ES)==true;
                if (es || FaultSkipWaitES) { _bus.Write(SignalOwner.Vehicle,LptSignal.BUSY,true); State=VehState.Acting; _t=now; }
                break;
            case VehState.Acting:
                if (now-_t > 100) // 模拟动作耗时
                {
                    if (!FaultNoComplete)
                    {
                        if (_dir==TransferDirection.Deposit) _phy.DockCarrier();   // 放到口上
                        else _phy.UndockCarrier();                                 // 从口取走
                        _bus.Write(SignalOwner.Vehicle,LptSignal.COMPT,true);
                        State=VehState.Done;
                    }
                }
                break;
            case VehState.Done:
                if (_bus.TryRead(LptSignal.ES)==false)
                { _bus.Write(SignalOwner.Vehicle,LptSignal.COMPT,false);
                  _bus.Write(SignalOwner.Vehicle,LptSignal.BUSY,false);
                  _bus.Write(SignalOwner.Vehicle,LptSignal.TR_REQ,false); State=VehState.Leave; }
                break;
        }
    }
}
```

### 仿真器与 Deposit 测试
```csharp
public sealed class HandshakeSimulator
{
    public LptSignalBus Bus { get; }
    public LoadPortFsm Port { get; }
    public VehicleFsm Veh { get; }
    private long _now;
    public HandshakeSimulator(IPortPhysical phy)
    { Bus=new(); Port=new(Bus,phy); Veh=new(Bus,phy); Bus.Write(SignalOwner.Shared,LptSignal.VALID,true); }
    public void Run(TransferDirection dir, int ticks=200)
    {
        Port.SetReady(true); Veh.Begin(dir,0);
        for (int i=0;i<ticks;i++){ _now+=10; Bus.Tick(10); Port.Tick(_now); Veh.Tick(_now);
            if (Port.State==PortState.Idle && i>2) break; }
    }
}
public class E84Tests
{
    [Fact] public void Deposit_正常_口上有盒且回Idle()
    {
        var phy=new InMemoryPort();                 // 起始空口
        var sim=new HandshakeSimulator(phy);
        sim.Run(TransferDirection.Deposit);
        Assert.True(phy.HasCarrier());
        Assert.Equal(PortState.Idle, sim.Port.State);
    }
    [Fact] public void Pickup_正常_口空且回Idle()
    {
        var phy=new InMemoryPort(); phy.DockCarrier(); // 起始有盒
        var sim=new HandshakeSimulator(phy);
        sim.Run(TransferDirection.Pickup);
        Assert.False(phy.HasCarrier());
        Assert.Equal(PortState.Idle, sim.Port.State);
    }
    [Fact] public void 小车不等ES_设备不推进()
    {
        var phy=new InMemoryPort();
        var sim=new HandshakeSimulator(phy){ };
        sim.Veh.FaultSkipWaitES=true;
        sim.Run(TransferDirection.Deposit);
        // 设备未许可, 交接不应正常完成
        Assert.NotEqual(PortState.Idle, sim.Port.State);
    }
    [Fact] public void T_compt超时_进安全保持()
    {
        var phy=new InMemoryPort();
        var sim=new HandshakeSimulator(phy);
        sim.Veh.FaultNoComplete=true;
        sim.Run(TransferDirection.Deposit, ticks:4000);
        Assert.Equal(PortState.SafeHold, sim.Port.State);
    }
}
```

## 四、今日易错点
1. 两侧都能写同一信号（违反唯一驱动源，必须所有者可写对方只读）。
2. VALID 无效仍返回电平（应返回 null/不可信，状态机冻结）。
3. 不去抖，信号一跳变就采信（毛刺误触发）。
4. 用真实线程导致时序不可复现（用逻辑步进时钟）。
5. 收到 TR_REQ 不查物理/逻辑一致性就置 ES（空口取、无许可取要拒绝）。
6. Deposit/Pickup 的 COMPT 完成判据写反（放=在位 ON，取=在位 OFF）。
7. 动作中超时回 Idle 重试（应 SafeHold 不自动重试）。
8. 故障注入只改一侧、忘了另一侧应不配合（互锁要靠双方验证）。
